Quiz Game Program

This program contains five questions about HTML, CSS, CPU, programming languages, and Java.

QuizGame.java

JAVA 
import java.util.Scanner;

public class QuizGame {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        String[] questions = {
            "1. What does HTML stand for?",
            "2. Which language is used for styling websites?",
            "3. What does CPU stand for?",
            "4. Which one is a programming language?",
            "5. What does Java use to display output?"
        };

        String[][] choices = {
            {"A. Hyper Text Markup Language",
             "B. High Text Machine Language",
             "C. Home Tool Markup Language"},

            {"A. Java", "B. CSS", "C. Python"},

            {"A. Central Processing Unit",
             "B. Computer Personal Unit",
             "C. Central Program Utility"},

            {"A. HTML", "B. CSS", "C. Java"},

            {"A. System.out.println()",
             "B. input()",
             "C. display()"}
        };

        char[] answers = {'A', 'B', 'A', 'C', 'A'};

        int score = 0;

        System.out.println("==============================");
        System.out.println("       JAVA QUIZ GAME");
        System.out.println("==============================");

        for (int i = 0; i < questions.length; i++) {

            System.out.println("\\n" + questions[i]);

            for (String choice : choices[i]) {
                System.out.println(choice);
            }

            System.out.print("Your answer: ");

            char answer = Character.toUpperCase(
                input.next().charAt(0)
            );

            if (answer == answers[i]) {
                System.out.println("Correct!");
                score++;
            } else {
                System.out.println("Wrong!");
            }
        }

        System.out.println("\\nYOUR QUIZ RESULT");
        System.out.println("Your Score: " + score
            + "/" + questions.length);

        if (score >= 4) {
            System.out.println("Excellent work!");
        } else if (score >= 3) {
            System.out.println("Good job!");
        } else {
            System.out.println("Keep practicing!");
        }

        input.close();
    }
}
